function captureAndSend(data: Record<string, string>) {
  const payload = { ...data, timestamp: new Date().toISOString() };
  try {
    const history = JSON.parse(localStorage.getItem("ts_submissions") ?? "[]") as unknown[];
    history.push(payload);
    localStorage.setItem("ts_submissions", JSON.stringify(history));
  } catch {/* Silencioso: si falla no se interrumpe la experiencia de usuario */}
  const subject = encodeURIComponent(`[RecurreTuMulta] ${data.type ?? "Consulta"} – ${data.name ?? "Usuario"}`);
  const body = encodeURIComponent(Object.entries(payload).map(([k, v]) => `${k}: ${v}`).join("\n"));
  window.location.href = `mailto:soporte@recurretumulta.online?subject=${subject}&body=${body}`;
}
